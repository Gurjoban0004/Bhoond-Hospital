        const form = document.getElementById("contactForm");
        const formMessage = document.getElementById("formMessage");

        function showMessage(msg, success) {
            formMessage.textContent = msg;
            formMessage.style.display = 'block';
            formMessage.className = 'form-message ' + (success ? 'success' : 'error');
            formMessage.style.background = success ? '#d4edda' : '#fdecea';
            formMessage.style.color = success ? '#155724' : '#7a1f1f';
            setTimeout(() => formMessage.style.display = 'none', 4500);
        }

        form.addEventListener("submit", async function (e) {
            e.preventDefault();
            const name = form.name.value.trim();
            const email = form.email.value.trim();
            const subject = form.subject.value.trim();
            const message = form.message.value.trim();
            const phone = form.phone.value.trim();

            const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            if (!name || !email || !subject || !message) {
                showMessage("Please complete all required fields.", false);
                return;
            }
            if (!emailRegex.test(email)) {
                showMessage("Please enter a valid email address.", false);
                return;
            }

            const submitBtn = form.querySelector('button[type="submit"]');
            const original = submitBtn.textContent;
            submitBtn.textContent = 'Sending...';
            submitBtn.disabled = true;

            try {
                const response = await fetch('/api/contacts', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ name, email, phone, subject, message }),
                });

                const result = await response.json();

                if (response.ok) {
                    showMessage(result.message || "Thank you — your message has been received. We'll respond shortly.", true);
                    form.reset();
                } else {
                    showMessage(result.error || "Something went wrong. Please try again later.", false);
                }
            } catch (err) {
                showMessage("Could not reach the server. Please try again later.", false);
            } finally {
                submitBtn.disabled = false;
                submitBtn.textContent = original;
            }
        });
    </script>

    <script>
        // ── Panel switching ───────────────────────────────────────────────────
        const panels = {
            login:  { tab: 'tabLogin',  panel: 'panelLogin'  },
            signup: { tab: 'tabSignup', panel: 'panelSignup' },
        };
        function showPanel(name) {
            Object.keys(panels).forEach(k => {
                const t = document.getElementById(panels[k].tab);
                const p = document.getElementById(panels[k].panel);
                const active = k === name;
                t.classList.toggle('active', active);
                t.setAttribute('aria-selected', String(active));
                active ? p.removeAttribute('hidden') : (p.hidden = true);
            });
            setTimeout(() => {
                const first = document.querySelector('#' + panels[name].panel + ' input');
                if (first) first.focus();
            }, 120);
        }

        // ── Helpers ───────────────────────────────────────────────────────────

            if (!inp) return;
            inp.type = inp.type === 'password' ? 'text' : 'password';
            const el = document.getElementById(toggleId);
            if (el) el.style.opacity = inp.type === 'text' ? '0.9' : '0.7';
        }
        function setError(id, msg) {
            const el = document.getElementById(id);
            if (el) el.textContent = msg || '';
        }
        function setNotice(id, msg, type = 'success') {
            const el = document.getElementById(id);
            if (!el) return;
            el.innerHTML = msg
                ? `<div class="notice ${type === 'success' ? 'success' : 'error'}">${msg}</div>`
                : '';
        }
        function setBtnLoading(btn, loading, label) {
            btn.disabled = loading;
            btn.textContent = loading ? label : btn.dataset.original;
        }

        // ── Password strength ─────────────────────────────────────────────────
        function scorePassword(pw) {
            if (!pw) return 0;
            let score = 0;
            if (pw.length >= 8) score++;
            if (/[A-Z]/.test(pw)) score++;
            if (/[0-9]/.test(pw)) score++;
            if (/[@$!%*?&]/.test(pw)) score++;
            return Math.min(score, 4);
        }
        const strengthBar = document.getElementById('strengthBar');
        const pwInput = document.getElementById('signupPassword');
        if (pwInput) {
            pwInput.addEventListener('input', () => {
                const s = scorePassword(pwInput.value);
                if (strengthBar) {
                    strengthBar.style.width = (s * 25) + '%';
                    strengthBar.style.background = s <= 1 ? '#ff6b6b' : s === 2 ? '#f6c343' : s === 3 ? '#7ed957' : '#05a284';
                }
            });
        }

        // ── LOGIN ─────────────────────────────────────────────────────────────
        async function onLogin(e) {
            e.preventDefault();
            const email    = document.getElementById('loginEmail').value.trim();
            const password = document.getElementById('loginPassword').value;

            setError('loginEmailError', '');
            setError('loginPasswordError', '');
            setNotice('loginNotice', '');

            if (!email)    { setError('loginEmailError', 'Enter your email.'); return false; }
            if (!password) { setError('loginPasswordError', 'Enter your password.'); return false; }

            const btn = document.getElementById('loginBtn');
            btn.dataset.original = btn.textContent;
            setBtnLoading(btn, true, 'Signing in...');

            try {
                const res  = await fetch('/api/login', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ email, password }),
                });
                const data = await res.json();

                if (res.ok) {
                    // Save user info to localStorage so the site knows who's logged in
                    localStorage.setItem('bhoond_user', JSON.stringify(data.user));
                    setNotice('loginNotice', `Welcome back, ${data.user.name}! Redirecting...`, 'success');
                    setTimeout(() => { window.location.href = '../index.html'; }, 900);
                } else {
                    setNotice('loginNotice', data.error || 'Login failed.', 'error');
                }
            } catch {
                setNotice('loginNotice', 'Could not reach server. Please try again.', 'error');
            } finally {
                setBtnLoading(btn, false);
            }
            return false;
        }

        // ── SIGN UP ───────────────────────────────────────────────────────────
        async function onSignup(e) {
            e.preventDefault();
            const name     = document.getElementById('fullName').value.trim();
            const email    = document.getElementById('signupEmail').value.trim();
            const password = document.getElementById('signupPassword').value;
            const confirm  = document.getElementById('confirmPassword').value;

            setError('nameError', '');
            setError('signupEmailError', '');
            setError('signupPasswordError', '');
            setError('confirmPasswordError', '');
            setNotice('signupNotice', '');

            let ok = true;
            if (!name || name.length < 3)          { setError('nameError', 'Full name must be at least 3 characters.'); ok = false; }
            if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) { setError('signupEmailError', 'Enter a valid email.'); ok = false; }
            if (password.length < 8)               { setError('signupPasswordError', 'Password must be at least 8 characters.'); ok = false; }
            if (password !== confirm)              { setError('confirmPasswordError', 'Passwords do not match.'); ok = false; }
            if (!ok) return false;

            const btn = document.querySelector('#panelSignup button[type="submit"]');
            btn.dataset.original = btn.textContent;
            setBtnLoading(btn, true, 'Creating...');

            try {
                const res  = await fetch('/api/signup', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ name, email, password }),
                });
                const data = await res.json();

                if (res.ok) {
                    setNotice('signupNotice', 'Account created! Check your email for a welcome message. Redirecting to login...', 'success');
                    setTimeout(() => showPanel('login'), 2000);
                } else {
                    setNotice('signupNotice', data.error || 'Signup failed.', 'error');
                }
            } catch {
                setNotice('signupNotice', 'Could not reach server. Please try again.', 'error');
            } finally {
                setBtnLoading(btn, false);
            }
            return false;
        }

        function onResetPassword(e) {
            e && e.preventDefault();
            setNotice('loginNotice', 'Password reset is not yet available. Please contact support.', 'error');
        }
        function showHelp() {
            alert('For support, contact hospital IT at support@bhoondhospital.com or call +91 78373 30004.');
        }

            showPanel('login');
            document.getElementById('loginEmail').focus();
        });
